import React, { useState } from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';

const keys = ['C','⌫','%','÷','7','8','9','×','4','5','6','−','1','2','3','+','0','.','='];
const isOp = k => ['%','÷','×','−','+'].includes(k);

function calc(x, o, y) {
  if (o === '÷' && y === 0) return 'Error';
  const r = o === '+' ? x + y : o === '−' ? x - y : o === '×' ? x * y : o === '÷' ? x / y : x % y;
  return String(parseFloat(r.toFixed(8)));
}

export default function App() {
  const [cur, setCur] = useState('0');
  const [a, setA] = useState(0);
  const [op, setOp] = useState(null);
  const [fresh, setFresh] = useState(false);
  const [top, setTop] = useState('');

  const press = k => {
    if (/\d/.test(k)) {
      if (!op && fresh) setTop('');
      setCur(fresh || cur === '0' || cur === 'Error' ? k : cur + k);
      setFresh(false);
    }
    else if (k === '.') {
      if (fresh || cur === 'Error') {
        if (!op) setTop('');
        setCur('0.');
        setFresh(false);
      }
      else if (!cur.includes('.')) setCur(cur + '.');
    }
    else if (k === 'C') {
      setCur('0');
      setOp(null);
      setFresh(false);
      setTop('');
    }
    else if (k === '⌫') {
      if (fresh && op) return;
      setCur(cur.length > 1 && cur !== 'Error' && !(fresh && !op) ? cur.slice(0, -1) : '0');
    }
    else if (k === '=') {
      if (op && !fresh) {
        setTop(`${a} ${op} ${cur} =`);
        setCur(calc(a, op, parseFloat(cur)));
        setOp(null);
        setFresh(true);
      }
    }
    else if (isOp(k)) {
      const x = op && !fresh ? calc(a, op, parseFloat(cur)) : cur;
      setTop('');
      if (x === 'Error') {
        setCur(x);
        setOp(null);
        setFresh(true);
        return;
      }
      setCur(x);
      setA(parseFloat(x));
      setOp(k);
      setFresh(true);
    }
  };

  const main = op ? `${a} ${op} ${fresh ? '' : cur}` : cur;

  return (
    <View style={s.root}>
      <View style={s.screen}>
        <Text style={s.top} numberOfLines={1}>{top}</Text>
        <Text style={s.display} numberOfLines={1} adjustsFontSizeToFit>
          {main}
        </Text>
      </View>

      <View style={s.grid}>
        {keys.map(k => (
          <View key={k} style={[s.cell, k === '0' && { width: '50%' }]}>
            <TouchableOpacity
              style={[
                s.btn,
                isOp(k) || k === '=' ? s.orange :
                k === 'C' || k === '⌫' || k === '%' ? s.grey : s.dark
              ]}
              onPress={() => press(k)}
              activeOpacity={0.7}
            >
              <Text style={[
                s.txt,
                isOp(k) || k === '=' ? s.operatorText : null
              ]}>
                {k}
              </Text>
            </TouchableOpacity>
          </View>
        ))}
      </View>
    </View>
  );
}

const s = StyleSheet.create({
  root: {
    flex: 1,
    backgroundColor: '#171923',
    justifyContent: 'flex-end',
    paddingTop: 40,
    paddingHorizontal: 8,
    paddingBottom: 12,
  },
  screen: {
    backgroundColor: '#202331',
    borderRadius: 24,
    paddingTop: 25,
    paddingBottom: 15,
    marginBottom: 15,
    paddingHorizontal: 12,
  },
  top: {
    color: '#9CA3AF',
    fontSize: 20,
    textAlign: 'right',
    paddingHorizontal: 12,
    minHeight: 28,
  },
  display: {
    color: '#FFFFFF',
    fontSize: 50,
    fontWeight: '600',
    textAlign: 'right',
    paddingHorizontal: 12,
    paddingTop: 10,
    paddingBottom: 12,
  },
  grid: {
    flexDirection: 'row',
    flexWrap: 'wrap',
  },
  cell: {
    width: '25%',
    padding: 5,
  },
  btn: {
    height: 68,
    borderRadius: 22,
    alignItems: 'center',
    justifyContent: 'center',
  },
  txt: {
    color: '#FFFFFF',
    fontSize: 25,
    fontWeight: '500',
  },
  orange: {
    backgroundColor: '#FF9F43',
  },
  operatorText: {
    color: '#171923',
    fontWeight: 'bold',
  },
  grey: {
    backgroundColor: '#454B5C',
  },
  dark: {
    backgroundColor: '#292D3E',
  },
});
